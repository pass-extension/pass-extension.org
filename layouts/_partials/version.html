{{- /* Latest GitHub release of a repo URL, as {tag, url}, fetched at build time. Empty on any failure. */ -}} {{- $release := "" -}} {{- with try (resources.GetRemote (printf "%s/releases/latest" (replace . "https://github.com/" "https://api.github.com/repos/"))) -}} {{- with .Value -}} {{- with . | transform.Unmarshal -}} {{- $release = dict "tag" .tag_name "url" .html_url -}} {{- end -}} {{- end -}} {{- end -}} {{- return $release -}}
